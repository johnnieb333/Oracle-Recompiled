cmake_minimum_required(VERSION 3.20)
project(OracleLauncher LANGUAGES CXX)
add_executable(oracle-launcher WIN32 main.cpp)
target_compile_features(oracle-launcher PRIVATE cxx_std_17)
target_compile_definitions(oracle-launcher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(oracle-launcher PRIVATE comdlg32 bcrypt gdi32 user32)
if(MINGW)
  target_link_options(oracle-launcher PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
set_target_properties(oracle-launcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
